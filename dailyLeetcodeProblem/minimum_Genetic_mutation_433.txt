class Solution {
public:
    int minMutation(string startGene, string endGene, vector<string>& bank) {
        int ans = 0;
        set<string> s;
        for (int i = 0; i < bank.size(); i++) {
            s.insert(bank[i]);
        }

        queue<string> q;
        map<string, bool> visited;

        q.push(startGene);
        visited[startGene] = true;

        while (!q.empty()) {

            int qSize=q.size();

            while(qSize--){
              string curr = q.front();
            q.pop();

            // check
            if (curr == endGene) {
                return ans;
            }

            // generate neighbours
            for (int i = 0; i < 8; i++) {
                string temp = curr;
                for (int j = 0; j < 4; j++) {
                    if (j == 0) {
                        temp[i] = 'A';
                        if (!visited[temp] && s.contains(temp)) {
                            visited[temp] = true;
                            q.push(temp);
                             
                        }
                    } else if (j == 1) {
                        temp[i] = 'C';
                        if (!visited[temp] && s.contains(temp)) {
                            visited[temp] = true;
                            q.push(temp);
                             
                        }
                    } else if (j == 2) {
                        temp[i] = 'G';
                        if (!visited[temp] && s.contains(temp)) {
                            visited[temp] = true;
                            q.push(temp);
                            
                        }
                    } else if (j == 3) {
                        temp[i] = 'T';
                        if (!visited[temp] && s.contains(temp)) {
                            visited[temp] = true;
                            q.push(temp);
                            
                        }
                    }
                }
            }

            }

            ans++;

            
        }

        return -1;
    }
};
